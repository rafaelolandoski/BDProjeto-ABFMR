    %% =====================================================
    %% HERANÇA (PESSOA -> especializações)
    %% Pessoa pode ou não ser cada especialização (0..1)
    %% =====================================================
    PESSOA ||--o| PALESTRANTE : "é um"
    PESSOA ||--o| PARTICIPANTE : "é um"
    PESSOA ||--o| INSTRUTOR : "é um"
    PESSOA ||--o| ORGANIZADOR : "é um"

    PESSOA {
        char11 CPF PK
        varchar50 nome "NOT NULL"
        varchar50 email UK "nullable"
        boolean e_palestrante "NOT NULL DEFAULT FALSE"
        boolean e_organizador "NOT NULL DEFAULT FALSE"
        boolean e_participante "NOT NULL DEFAULT FALSE"
        boolean e_instrutor "NOT NULL DEFAULT FALSE"
    }

    PALESTRANTE {
        char11 CPF PK, FK
        varchar30 especialidade
        text biografia "DEFAULT NULL"
    }

    PARTICIPANTE {
        char11 CPF PK, FK
    }

    INSTRUTOR {
        char11 CPF PK, FK
        text ajuda "DEFAULT 'geral'"
    }

    ORGANIZADOR {
        char11 CPF PK, FK
        varchar30 cargo
    }

    %% =====================================================
    %% ENTIDADES PRINCIPAIS
    %% =====================================================
    EVENTO {
        integer id_evento PK
        varchar50 nome UK "UNIQUE(nome, edicao)"
        bigint edicao UK "UNIQUE(nome, edicao)"
        date data_inicio "CHECK inicio <= fim"
        date data_fim
    }

    ATIVIDADE {
        integer id_atividade PK
        integer id_evento FK, UK "UNIQUE(titulo, id_evento)"
        varchar50 titulo UK "UNIQUE(titulo, id_evento)"
        text descricao
    }

    ESPACO {
        integer id_espaco PK
        varchar50 nome
        smallint capacidade "CHECK > 0"
    }

    SESSAO {
        smallint n_sessao PK
        integer id_atividade PK, FK
        integer id_espaco FK
        timestamp horario_inicio "CHECK inicio < fim"
        timestamp horario_fim
        smallint vagas "CHECK > 0 OR NULL (NULL = sem limite)"
    }

    TIPO_RECURSO {
        integer id_tipo PK
        proposito tipo UK "ENUM"
    }

    RECURSO {
        integer id_recurso PK
        integer id_tipo FK
        integer id_espaco FK
        varchar50 nome
        varchar50 marca
        numeric preco "numeric(9,2)"
        date dataCompra
        integer validade "anos, CHECK > 0"
    }

    ORGANIZACAO {
        char14 CNPJ PK
        varchar50 nome
        tp org_tipo "ENUM"
    }

    %% =====================================================
    %% TABELAS ASSOCIATIVAS (M:N)
    %% =====================================================
    ORG_EVENTO {
        char14 CNPJ PK, FK
        integer id_evento PK, FK
    }

    PART_EVENTO {
        integer id_evento PK, FK
        char11 CPF PK, FK
    }

    PART_SESSAO {
        char11 CPF PK, FK
        smallint n_sessao PK, FK
        integer id_atividade PK, FK
    }

    LISTA_DE_ESPERA {
        char11 CPF PK, FK
        smallint n_sessao PK, FK
        integer id_atividade PK, FK
        timestamp dataInscricao "DEFAULT CURRENT_TIMESTAMP"
    }

    INST_SESSAO {
        char11 CPF PK, FK
        smallint n_sessao PK, FK
        integer id_atividade PK, FK
    }

    ATIV_TIPO_RECURSO {
        integer id_atividade PK, FK
        integer id_tipo PK, FK
        smallint quantidade "DEFAULT 1, CHECK > 0"
    }

    PALES_ATIV {
        char11 CPF PK, FK
        integer id_atividade PK, FK
    }

    ORGANIZADOR_ORGANIZACAO {
        char11 CPF PK, FK
        char14 CNPJ PK, FK
    }

    %% =====================================================
    %% RELACIONAMENTOS 1:N
    %% (-- identificador: FK faz parte da PK | .. não identificador)
    %% =====================================================
    EVENTO ||..o{ ATIVIDADE : "possui"
    ATIVIDADE ||--o{ SESSAO : "é realizada em"
    ESPACO ||..o{ SESSAO : "sedia"
    ESPACO ||..o{ RECURSO : "contém"
    TIPO_RECURSO ||..o{ RECURSO : "classifica"

    %% =====================================================
    %% LIGAÇÕES DAS TABELAS ASSOCIATIVAS
    %% =====================================================
    ORGANIZACAO ||--o{ ORG_EVENTO : "organiza"
    EVENTO ||--o{ ORG_EVENTO : "é organizado por"

    PARTICIPANTE ||--o{ PART_EVENTO : "inscreve-se"
    EVENTO ||--o{ PART_EVENTO : "tem inscritos"

    PARTICIPANTE ||--o{ PART_SESSAO : "assiste"
    SESSAO ||--o{ PART_SESSAO : "tem participantes"

    PARTICIPANTE ||--o{ LISTA_DE_ESPERA : "aguarda"
    SESSAO ||--o{ LISTA_DE_ESPERA : "tem fila de espera"

    INSTRUTOR ||--o{ INST_SESSAO : "auxilia"
    SESSAO ||--o{ INST_SESSAO : "tem instrutores"

    ATIVIDADE ||--o{ ATIV_TIPO_RECURSO : "requer"
    TIPO_RECURSO ||--o{ ATIV_TIPO_RECURSO : "é requerido por"

    PALESTRANTE ||--o{ PALES_ATIV : "apresenta"
    ATIVIDADE ||--o{ PALES_ATIV : "tem palestrantes"

    ORGANIZADOR ||--o{ ORGANIZADOR_ORGANIZACAO : "faz parte de"
    ORGANIZACAO ||--o{ ORGANIZADOR_ORGANIZACAO : "tem organizadores"
